# LivingDoc 后端依赖
#
# 安装：
#   python -m venv .venv
#   .\.venv\Scripts\python.exe -m pip install -r requirements.txt
#
# 网络慢可以加国内镜像：
#   ... -m pip install -r requirements.txt -i https://pypi.tuna.tsinghua.edu.cn/simple
#
# 版本策略：用 ">=x,<y" 而不是死锁 "==x.y.z"。
# 大作业要在别人的机器上跑得起来，锁太死反而会因 Python 版本不同装不上；
# 真正的可复现性靠 requirements.lock.txt（pip freeze 生成，见文末说明）。

pydantic>=2.7,<3.0           # 数据校验（FastAPI 的底层）

# ---------- 异步任务 ----------

# broker/result backend 的两种实现：

# ---------- 测试 ----------

tree-sitter>=0.24,<0.27
